cmake_minimum_required(VERSION 3.20)
project(sort_lab VERSION 0.2.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

add_executable(sort_lab src/sort_lab.cpp)
target_include_directories(sort_lab PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/include)
if(MSVC)
  target_compile_options(sort_lab PRIVATE /W4 /permissive-)
else()
  target_compile_options(sort_lab PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
endif()

include(CTest)
if(BUILD_TESTING)
  add_test(NAME sort_self_test COMMAND sort_lab --self-test)
  add_test(NAME sort_smoke_benchmark COMMAND sort_lab --trials 2 --warmups 0 --sizes 0,1,32,257 --patterns random,all_equal,organ_pipe --algorithms insertion,heap,merge,quick_3way,intro,radix_lsd,std_sort)
  find_package(Python3 COMPONENTS Interpreter QUIET)
  if(Python3_Interpreter_FOUND)
    add_test(NAME analysis_self_test COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tools/analyze.py --self-test)
  endif()
endif()
